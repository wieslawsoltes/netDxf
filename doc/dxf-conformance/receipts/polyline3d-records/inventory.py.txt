from pathlib import Path
from collections import Counter
import io,json,hashlib
from ezdxf.lldxf.tagger import ascii_tags_loader,binary_tags_loader
root=Path('/workspace/scratch/2ec4aa26f01f'); sources=root/'ucs-base-research/libredwg'
counts=Counter();profiles=Counter();flags=Counter();markers=Counter();examples=[];files=[];failures=[]
for path in sorted(sources.glob('*.dxf')):
 data=path.read_bytes()
 try:
  raw=list(binary_tags_loader(data) if data.startswith(b'AutoCAD Binary DXF') else ascii_tags_loader(io.StringIO(data.decode('latin1'),newline=None)))
  rows=[];row=[]
  for t in raw:
   if t.code==0 and row:rows.append(row);row=[]
   row.append((t.code,t.value))
  if row:rows.append(row)
  profile=next((raw[i+1].value for i,t in enumerate(raw[:-1]) if t.code==9 and t.value=='$ACADVER'),None)
  summary=Counter();metadata=[]
  for row in rows:
   if row[0][1] not in ('VERTEX','SEQEND'):continue
   kind=row[0][1];summary[kind]+=1;counts[kind]+=1;profiles[(profile,kind)]+=1
   subclass=None;group=None;own=[];reactors=[];extension=[];common=[];xdata=[]
   for c,v in row[1:]:
    if c==100:subclass=v
    if c==102:group=None if v=='}' else v;continue
    if c==1001:xdata.append(v)
    if c==330 and subclass is None and group is None:own.append(v)
    if group=='{ACAD_REACTORS' and c==330:reactors.append(v)
    if group=='{ACAD_XDICTIONARY' and c==360:extension.append(v)
    if subclass=='AcDbEntity' and c not in (100,8):common.append(c)
   handle=next((v for c,v in row if c==5),None)
   if kind=='VERTEX':
    flags[str(next((int(v) for c,v in row if c==70),0))]+=1
    markers[tuple(v for c,v in row if c==100)]+=1
   for label,value in [('identity',handle),('owner',own),('reactors',reactors),('extension',extension),('xdata',xdata),('common_overrides',common)]:
    if value:counts[kind+'/'+label]+=1;summary[kind+'/'+label]+=1
   if reactors or extension or xdata or common:
    metadata.append({'handle':handle,'kind':kind,'owner':own,'reactors':reactors,'extension':extension,'xdata':xdata,'common_override_codes':common,'packet':[[c,{'bytes':v.hex()} if isinstance(v,bytes) else v] for c,v in row]})
  files.append({'file':str(path.relative_to(root)),'sha256':hashlib.sha256(data).hexdigest(),'profile':profile,'summary':dict(summary),'metadata':metadata})
 except Exception as e:failures.append({'file':str(path.relative_to(root)),'error':str(e)})
result={'summary':dict(counts),'profiles':{str(k):v for k,v in profiles.items()},'vertex_flags':dict(flags),'vertex_subclasses':{str(k):v for k,v in markers.items()},'files':files,'failures':failures}
(root/'vertex-record-research/inventory.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='files'},indent=2))
